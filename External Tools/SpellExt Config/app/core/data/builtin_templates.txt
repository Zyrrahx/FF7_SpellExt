; FF7 SpellExt Config - the built-in templates. This file is the config tool's; the DLL does not read it.
; The format is the one of FF7_SpellExt_Templates.txt (the user's): see the tool's README, "Templates".
;
; Every template comes from the game's own data only: the bases and their names from KERNEL.BIN (read
; 2026-10-27/28), the values and words from the readmes and the readers. Optional keys are commented out, so the
; user turns them on; testing aids are always commented out. A template proves only that the DLL would log nothing
; new: what the thing does in play is the readme's word, not seen in the game.
;
; Placeholders: NAME, type, default, label. A default with commas is written in double quotes.

; ==== Items (FF7_SpellExt_Items.ini) ===============================================================================

[template item_restore_hp]
title = Restore HP item
kind = item
file = FF7_SpellExt_Items.ini
note = Copies the Potion (kernel item 0). In battle it restores 20 HP per point of strength (a Potion's 5 restores 100), so strength is HP / 20, 255 at most; menu_hp is what it restores from the Item menu.
note = With HP = all, strength is 255: in battle it then restores 20 x 255 = 5100 HP, not all.
placeholder = ID, id
placeholder = NAME, text31, Tonic, Name
placeholder = DESC, text95, Restores HP, Description
placeholder = HP, int(0..9999)|all, 500, HP restored in the menu
placeholder = PRICE, int(0..2147483647), 300, Price
derived = STRENGTH = min(255, HP / 20); all = 255
lines
    [<ID>]
    name = <NAME>
    description = <DESC>
    base = 0                ; Potion: its use in battle, with the strength below
    strength = <STRENGTH>   ; a Potion's is 5 and restores 100 HP: 20 HP per point, 255 at most
    menu_hp = <HP>
    price = <PRICE>
    ;give_on_load = yes     ; testing aid: one in the stock whenever a save is loaded; remove when done
